# Runtime dependencies for services/api — Phase 1.
#
# Exact pins, not ranges: a hackathon demo that worked yesterday and breaks
# today because a transitive release changed is not worth the convenience.
# Regenerate with `pip freeze` only after a deliberate upgrade.
#
# Nothing here is speculative. Libraries for later phases (GeoPandas, scikit-
# learn, XGBoost) are added in the phase that first uses them, per the project
# rule against carrying dependencies for appearance.

# --- web framework -----------------------------------------------------------
# FastAPI for the API surface, Uvicorn to serve it. The [standard] extra brings
# uvloop, httptools and websockets — the last is required by Phase 9 live
# tracking, so it is not optional weight.
fastapi==0.115.12
uvicorn[standard]==0.34.2

# --- validation and settings -------------------------------------------------
# Pydantic v2 models are the request/response contract and the OpenAPI schema.
# pydantic-settings loads .env into the typed Settings object in app/core.
pydantic==2.11.4
pydantic-settings==2.9.1
# Validates EmailStr on user creation instead of a hand-rolled regex.
email-validator==2.2.0

# --- database ----------------------------------------------------------------
# SQLAlchemy 2.0 declarative models, async in the request path.
sqlalchemy==2.0.41
# PostGIS column types (Geometry) and the spatial functions used by scoped
# queries and hazard intersection.
GeoAlchemy2==0.17.1
# Schema versioning. The initial revision is frozen, not autogenerated.
alembic==1.15.2
# asyncpg drives the async engine used by request handlers.
asyncpg==0.30.0
# psycopg 3 drives the sync engine used by Alembic and scripts/seed.py, which
# have no event loop and should not pretend to.
psycopg[binary]==3.2.9

# --- cache, queues, realtime -------------------------------------------------
# Redis backs rate limiting, WebSocket fan-out between workers, and cached
# provider responses so a weather lookup is not repeated per request.
redis==6.1.0

# --- security ----------------------------------------------------------------
# bcrypt directly rather than passlib: passlib's bcrypt backend reads a private
# attribute removed in bcrypt 4.1 and we need exactly two functions from it.
bcrypt==4.3.0
# JWT signing and verification, with every check enabled explicitly.
PyJWT==2.10.1
# Required by FastAPI to parse multipart bodies — field-report photo upload.
python-multipart==0.0.20

# --- outbound HTTP -----------------------------------------------------------
# One async client for every provider adapter: OSRM, Open-Meteo, Ollama.
httpx==0.28.1
